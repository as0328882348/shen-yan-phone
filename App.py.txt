chat_history = [{"role": "system", "content": SYSTEM_PROMPT}]

def ask_shen_yan(user_text: str) -> str:
    chat_history.append({"role": "user", "content": user_text})
    if len(chat_history) > 31:
        del chat_history[1:3]

    response = client.chat.completions.create(
        model="deepseek-chat",
        messages=chat_history,
        temperature=1.3,
        max_tokens=1000
    )
    reply = response.choices[0].message.content
    chat_history.append({"role": "assistant", "content": reply})
    return reply

class ChatRequest(BaseModel):
    message: str

@app.get("/", response_class=HTMLResponse)
async def get_phone_ui():
    with open("index.html", "r", encoding="utf-8") as f:
        return f.read()

@app.post("/api/chat")
async def web_chat(req: ChatRequest):
    return {"reply": ask_shen_yan(req.message)}

@app.post("/callback")
async def line_callback(request: Request):
    signature = request.headers.get("X-Line-Signature", "")
    body = await request.body()
    try:
        handler.handle(body.decode("utf-8"), signature)
    except InvalidSignatureError:
        raise HTTPException(status_code=400, detail="Invalid signature")
    return "OK"

@handler.add(MessageEvent, message=TextMessage)
def handle_line_message(event):
    if LINE_CHANNEL_ACCESS_TOKEN:
        reply_text = ask_shen_yan(event.message.text)
        line_bot_api.reply_message(event.reply_token, TextSendMessage(text=reply_text))
